\begin{warnbox}[frametitle={Antes de empezar: parametrizaciones (fuente nº 1 de errores)}]
El Formulario 2 usa $\beta$ como \textbf{escala} ($=\E$) para Exponencial y
Gamma. Cuando un enunciado dice ``exponencial de parámetro $\lambda$'', ese
$\lambda$ es la \textbf{tasa}: $\beta=1/\lambda$ y la media es $1/\lambda$.
\[
\text{Exp}(\beta):\ \tfrac1\beta e^{-x/\beta},\ \E=\beta
\qquad\longleftrightarrow\qquad
\lambda e^{-\lambda x},\ \E=1/\lambda .
\]
Además: $N(\mu,\sigma^2)$ lleva la \textbf{varianza} (no la desviación) y la
Geométrica / Binomial negativa del formulario cuentan \textbf{ensayos} (no
fracasos). Regla: al leer ``parámetro $\lambda$'', anota al margen
``$\lambda=$ tasa, media $=1/\lambda$, $\beta=1/\lambda$''.
\end{warnbox}

\section{Variable aleatoria, densidad y acumulada}

\begin{defbox}[frametitle={Discreta: función de probabilidad}]
$p_X(x)=\Prob(X=x)\ge 0$, \quad
$\displaystyle\sum_{x\in S_X}p_X(x)=1$, \quad
$\Prob(X\in A)=\displaystyle\sum_{x\in A}p_X(x)$.

\smallskip
$S_X$ es el \textbf{soporte}: los $x$ con $\Prob(X=x)>0$.
\end{defbox}

\begin{defbox}[frametitle={Continua: función de densidad (fdp)}]
$f_X(x)\ge 0$, \quad
$\displaystyle\int_{-\infty}^{\infty}f_X(x)\,dx=1$, \quad
$\Prob(a<X\le b)=\displaystyle\int_a^b f_X(x)\,dx$.

\smallskip
Cada punto tiene probabilidad $0$: $\Prob(X=a)=0$.
\end{defbox}

\begin{defbox}[frametitle={Acumulada y cuantil}]
\[
F_X(x)=\Prob(X\le x),\qquad \Prob(a<X\le b)=F_X(b)-F_X(a).
\]
Continua: $F_X(x)=\displaystyle\int_{-\infty}^{x}f_X(t)\,dt$ y $f_X(x)=F_X'(x)$.
Discreta (escalera): $\Prob(X=x)=F(x)-F(x^-)$.

\smallskip
\textbf{Cuantil $p$:} el número $x_p$ con $F_X(x_p)=p$ (``área $p$ a la
izquierda''). La mediana es $x_{0.5}$; si $F$ es continua y creciente,
$x_p=F^{-1}(p)$.
\end{defbox}

\begin{warnbox}[frametitle={Dos trampas}]
\begin{itemize}
  \item $f_X(x)$ \textbf{no} es una probabilidad (puede superar 1); sólo las
        \emph{áreas} lo son.
  \item ``$<$'' vs ``$\le$'': da igual en continuas, \textbf{no} en discretas.
\end{itemize}
\end{warnbox}

\section{Esperanza, varianza, covarianza y sus propiedades}

\begin{defbox}[frametitle={Definiciones}]
\[
\E(X)=\sum_{x}x\,\Prob(X=x)=\int_{-\infty}^{\infty}x f_X(x)\,dx,
\]
\[
\E[g(X)]=\sum_{x}g(x)\Prob(X=x)=\int_{-\infty}^{\infty}g(x)f_X(x)\,dx,
\]
\[
\Var(X)=\E\big[(X-\E X)^2\big]=\E(X^2)-\big[\E(X)\big]^2,\qquad
\sigma_X=\sqrt{\Var(X)} .
\]
\[
\Cov(X,Y)=\E\big[(X-\E X)(Y-\E Y)\big]=\E(XY)-\E(X)\E(Y).
\]
\end{defbox}

\begin{thmbox}[frametitle={Propiedades de la esperanza}]
\begin{itemize}
  \item \textbf{Linealidad (siempre, sin independencia):}
        $\E(aX+b)=a\,\E(X)+b$;\quad $\E(X+Y)=\E(X)+\E(Y)$.
  \item General: $\E\!\big[\sum_{i=1}^n a_iX_i+b\big]=\sum_{i=1}^n a_i\,\E(X_i)+b$;
        $\E(c)=c$.
  \item \textbf{Independencia (producto):} $X\ind Y\Rightarrow
        \E(XY)=\E(X)\E(Y)$ y $\E[g(X)h(Y)]=\E[g(X)]\,\E[h(Y)]$.
  \item \textbf{No lineal:} $\E[g(X)]\ne g(\E X)$ salvo si $g$ es lineal.
        Si $g$ es convexa, $\E[g(X)]\ge g(\E X)$ \emph{(Jensen)}.
  \item \textbf{Esperanza total:} $\E(X)=\E\big[\E(X\mid Y)\big]$.
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Propiedades de la varianza}]
\begin{itemize}
  \item $\Var(X)=\E(X^2)-\big[\E(X)\big]^2\ge 0$;\quad $\Var(c)=0$.
  \item \textbf{Desplazamiento y escala:} $\Var(X+b)=\Var(X)$,\quad
        $\Var(aX+b)=a^2\,\Var(X)$.
  \item \textbf{Suma y resta:}
        $\Var(X+Y)=\Var(X)+\Var(Y)+2\Cov(X,Y)$;\quad
        $\Var(X-Y)=\Var(X)+\Var(Y)-2\Cov(X,Y)$.
  \item General: $\Var\!\big(\sum_i X_i\big)=\sum_i\Var(X_i)
        +2\sum_{i<j}\Cov(X_i,X_j)$.
  \item Si las $X_i$ son \textbf{independientes} (o sólo incorreladas):
        $\Var\!\big(\sum_i X_i\big)=\sum_i\Var(X_i)$.
  \item \textbf{Varianza total:}
        $\Var(X)=\E\big[\Var(X\mid Y)\big]+\Var\big(\E[X\mid Y]\big)$.
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Propiedades de la covarianza y correlación}]
\begin{itemize}
  \item $\Cov(X,Y)=\E(XY)-\E(X)\E(Y)=\Cov(Y,X)$;\quad $\Cov(X,X)=\Var(X)$.
  \item \textbf{Bilinealidad:}
        $\Cov(aX+b,\;cY+d)=ac\,\Cov(X,Y)$ (las constantes no afectan).
  \item $\Cov(X+Y,Z)=\Cov(X,Z)+\Cov(Y,Z)$;\quad
        $\Cov\!\big(\sum_i a_iX_i,\sum_j b_jY_j\big)=\sum_{i}\sum_{j}a_i b_j\,\Cov(X_i,Y_j)$.
  \item \textbf{Independencia:} $X\ind Y\Rightarrow \Cov(X,Y)=0$. El recíproco
        \textbf{no} es cierto en general (sí para la \textbf{normal bivariada}).
  \item $\rho=\Corr(X,Y)=\dfrac{\Cov(X,Y)}{\sqrt{\Var(X)\Var(Y)}}\in[-1,1]$;\quad
        $\Corr(aX+b,cY+d)=\operatorname{sign}(ac)\,\rho$.
  \item \textbf{Cauchy–Schwarz:} $\big[\Cov(X,Y)\big]^2\le\Var(X)\Var(Y)$.
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Desigualdades}]
\textbf{Markov} ($X\ge0$, $a>0$): $\Prob(X\ge a)\le\dfrac{\E(X)}{a}$.\qquad
\textbf{Chebyshev} ($k>0$):
\[
\Prob(|X-\mu|\ge k)\le\frac{\sigma^2}{k^2}
\qquad\Longleftrightarrow\qquad
\Prob(|X-\mu|\ge k\sigma)\le\frac{1}{k^2}.
\]
A lo más $1/4$ de la probabilidad está a $2\sigma$ o más, y a lo más $1/9$ a
$3\sigma$ o más, \textbf{sea cual sea la distribución}.
\end{thmbox}

\section{Función generadora de momentos (FGM)}

\begin{defbox}[frametitle={Definición y momentos}]
\[
M_X(t)=\E\big(e^{tX}\big),\qquad M_X(0)=1,\qquad
\frac{d^k}{dt^k}M_X(t)\Big|_{t=0}=\E(X^k),
\]
\[
\Var(X)=M_X''(0)-\big[M_X'(0)\big]^2 .
\]
\end{defbox}

\begin{thmbox}[frametitle={Propiedades y unicidad}]
\begin{itemize}
  \item \textbf{Transformación lineal:} $M_{aX+b}(t)=e^{bt}M_X(at)$.
  \item \textbf{Suma de independientes:} $U=X_1+\cdots+X_n$ con
        $X_i\ind\Rightarrow M_U(t)=\prod_{i=1}^n M_{X_i}(t)$; si son i.i.d.,
        $M_U(t)=[M_X(t)]^n$.
  \item \textbf{Unicidad:} si $M_X(t)=M_Y(t)$ para todo $t$ en un intervalo
        abierto que contiene a $0$, entonces $X$ y $Y$ tienen la \textbf{misma
        distribución}. \emph{Ésta es la clave del método FGM.}
\end{itemize}
\end{thmbox}

\begin{notebox}[frametitle={Receta del método FGM}]
(1) Escribe la FGM de cada sumando (tabla de la sección 4)
$\to$ (2) multiplícalas (usa $M_{a+bX}$ si hay constantes)
$\to$ (3) simplifica a la forma de una FGM conocida
$\to$ (4) identifica parámetros y concluye por unicidad.
\end{notebox}

\begin{itemize}

\section{Funciones Gamma y Beta (el truco del núcleo)}

\begin{defbox}[frametitle={Gamma}]
\[
\Gamma(\alpha)=\int_0^\infty x^{\alpha-1}e^{-x}\,dx,\ \alpha>0;\qquad
\Gamma(\alpha+1)=\alpha\,\Gamma(\alpha),\quad
\Gamma(n)=(n-1)!,\quad \Gamma(\tfrac12)=\sqrt\pi .
\]
\end{defbox}

\begin{defbox}[frametitle={Beta}]
\[
B(\alpha,\beta)=\int_0^1 x^{\alpha-1}(1-x)^{\beta-1}\,dx
=\frac{\Gamma(\alpha)\Gamma(\beta)}{\Gamma(\alpha+\beta)},\quad\alpha,\beta>0 .
\]
\end{defbox}

\begin{notebox}[frametitle={El truco del núcleo}]
Si reconoces el núcleo de una densidad conocida, la integral vale lo que le
falta para ser $1$:
\[
\int_0^\infty x^{\alpha-1}e^{-x/\beta}\,dx=\beta^\alpha\Gamma(\alpha),\quad
\int_0^1 x^{\alpha-1}(1-x)^{\beta-1}dx=B(\alpha,\beta),\quad
\int_{-\infty}^{\infty}e^{-\frac{(x-\mu)^2}{2\sigma^2}}dx=\sqrt{2\pi\sigma^2}.
\]
Momentos Gamma: $\E(X^k)=\beta^k\dfrac{\Gamma(\alpha+k)}{\Gamma(\alpha)}$.
\end{notebox}

\section{Distribuciones conjuntas}

\begin{defbox}[frametitle={Marginal, condicional e independencia}]
\[
f_X(x)=\int_{-\infty}^{\infty}f_{XY}(x,y)\,dy,\qquad
f_{X\mid Y}(x\mid y)=\frac{f_{XY}(x,y)}{f_Y(y)},\qquad
X\ind Y\iff f_{XY}(x,y)=f_X(x)f_Y(y).
\]
\[
\E[g(X,Y)]=\iint g(x,y)f_{XY}(x,y)\,dx\,dy,\quad
\E[X\mid Y=y]=\int x\,f_{X\mid Y}(x\mid y)\,dx,\quad
\E\big[\E(X\mid Y)\big]=\E(X).
\]
(En el caso discreto, cambia $\int$ por $\sum$.)
\end{defbox}

\begin{warnbox}[frametitle={Independencia: atajos}]
\begin{itemize}
  \item \textbf{Soporte no rectangular} (p. ej.\ $0<x<y<1$) $\Rightarrow$
        \textbf{no} son independientes.
  \item Soporte rectangular y $f_{XY}(x,y)=g(x)h(y)$ $\Rightarrow$ sí son
        independientes.
  \item $\Cov=0$ \textbf{no} implica independencia (salvo normal bivariada).
\end{itemize}
\end{warnbox}

\begin{notebox}[frametitle={Límites de integración (lee el soporte)}]
(1) \textbf{Dibuja} el soporte. (2) La variable de afuera (última que integras)
tiene límites \textbf{constantes}. (3) Para la de adentro, fija un valor de la de
afuera y mira dónde la recta \textbf{entra} y \textbf{sale} de la región.
(4) Si la frontera cambia de fórmula, \textbf{parte} la integral. (5) En una
marginal, la variable que no integras queda fija.
\end{notebox}

\section{Convergencias y ley de los grandes números}

\begin{defbox}[frametitle={Tipos de convergencia}]
\begin{itemize}
  \item \textbf{En probabilidad:} $X_n\pto X$ si
        $\displaystyle\lim_{n\to\infty}\Prob(|X_n-X|>\varepsilon)=0$.
  \item \textbf{En distribución:} $X_n\dto X$ si $\lim F_{X_n}(x)=F_X(x)$ en
        todo punto de continuidad de $F_X$ (basta que $M_{X_n}\to M_X$ cerca
        de $0$).
  \item \textbf{Casi segura:} $X_n\dcs X$ si $\Prob(\lim X_n=X)=1$. Es la más
        fuerte: c.s.\ $\Rightarrow$ en probabilidad $\Rightarrow$ en
        distribución.
\end{itemize}
\end{defbox}

\begin{thmbox}[frametitle={LGN y su prueba con Chebyshev}]
Si $X_1,\dots,X_n$ son i.i.d.\ con media $\mu$ finita, $\bar X_n\pto\mu$. Con
$\sigma^2<\infty$:
\[
\Prob(|\bar X_n-\mu|\ge\varepsilon)\le\frac{\sigma^2/n}{\varepsilon^2}
=\frac{\sigma^2}{n\varepsilon^2}\xrightarrow[n\to\infty]{}0 .
\]
\end{thmbox}

\section{Transformaciones de variables aleatorias \textmd{(Clase 1)}}

\begin{notebox}[frametitle={Método de la acumulada (el más general)}]
(1) Soporte de $U=h(Y)$. (2) $F_U(u)=\Prob(h(Y)\le u)$. (3) Despeja $Y$ dentro
del evento (si $h$ decrece, se \textbf{voltea} la desigualdad). (4) Calcula con
$F_Y$. (5) $f_U=F_U'$, con su soporte.
\end{notebox}

\begin{thmbox}[frametitle={Jacobiano, una variable ($h$ monótona)}]
\[
f_U(u)=f_Y\big[h^{-1}(u)\big]\left|\frac{d\,h^{-1}(u)}{du}\right|,\qquad
u\in h(\text{soporte de }Y).
\]
\end{thmbox}

\begin{thmbox}[frametitle={$h$ no monótona: suma sobre las ramas}]
\[
f_U(u)=\sum_{k:\ u\in h(\text{tramo }k)}
f_Y\big(h_k^{-1}(u)\big)\left|\frac{d\,h_k^{-1}(u)}{du}\right| .
\]
Caso $U=Y^2$:\quad
$f_U(u)=\dfrac{f_Y(\sqrt u)+f_Y(-\sqrt u)}{2\sqrt u}$, $u>0$ (cuenta cuántas
raíces caen en el soporte). Si $h$ tiene tramos planos, $U$ puede tener
\textbf{átomos}: usa sólo la acumulada.
\end{thmbox}

\begin{thmbox}[frametitle={Jacobiano bivariado}]
\[
f_{U_1,U_2}(u_1,u_2)=f_{Y_1,Y_2}\big(h_1^{-1},h_2^{-1}\big)\,|J|,\qquad
J=\det\begin{bmatrix}
\dfrac{\partial y_1}{\partial u_1} & \dfrac{\partial y_1}{\partial u_2}\\[8pt]
\dfrac{\partial y_2}{\partial u_1} & \dfrac{\partial y_2}{\partial u_2}
\end{bmatrix}.
\]
$J$ es el jacobiano de la \textbf{inversa} (las $y$ respecto de las $u$). Atajo:
$J=1\big/\det\big(\partial(u_1,u_2)/\partial(y_1,y_2)\big)$. Si sólo piden una
$U_1$, añade una variable \textbf{auxiliar} y luego marginaliza.
\end{thmbox}

\begin{thmbox}[frametitle={Caso $n$-dimensional}]
\[
g(\mathbf u)=f\big(\mathbf y(\mathbf u)\big)\,\Big|\det\frac{\partial \mathbf y}{\partial \mathbf u}\Big|,\qquad
\det\frac{\partial \mathbf y}{\partial \mathbf u}
=\det\!\Big(\partial y_i/\partial u_j\Big)_{i,j}.
\]
\end{thmbox}

\begin{warnbox}[frametitle={Errores que hay que evitar}]
\begin{itemize}
  \item Olvidar el \textbf{soporte} de $U$ o el \textbf{valor absoluto} (sale
        densidad negativa).
  \item Usar el jacobiano directo en vez del de la inversa, o no voltear la
        desigualdad.
  \item Aplicar el Jacobiano cuando $h$ no es inyectiva (hay que sumar ramas).
  \item Límites de integración fijos cuando la región depende de $u$: dibuja
        primero.
\end{itemize}
\end{warnbox}

\section{Estadísticos de orden \textmd{(Clase 2)}}

$Y_{(1)}\le\cdots\le Y_{(n)}$: las $Y_1,\dots,Y_n$ i.i.d.\ ordenadas.
$Y_{(1)}=\min$, $Y_{(n)}=\max$.

\begin{thmbox}[frametitle={Máximo y mínimo}]
\[
F_{Y_{(n)}}(y)=[F(y)]^n,\qquad g_{(n)}(y)=n\,[F(y)]^{n-1}f(y),
\]
\[
F_{Y_{(1)}}(y)=1-[1-F(y)]^n,\qquad g_{(1)}(y)=n\,[1-F(y)]^{n-1}f(y).
\]
\end{thmbox}

\begin{thmbox}[frametitle={El $i$-ésimo ordenado (Teorema 1)}]
\[
f_{Y_{(i)}}(x)=\frac{n!}{(i-1)!\,(n-i)!}\,[F(x)]^{i-1}\,f(x)\,[1-F(x)]^{n-i},\qquad x\in\R .
\]
\end{thmbox}

\begin{thmbox}[frametitle={Densidades conjuntas}]
Todos a la vez ($y_1<\cdots<y_n$):
\[
f_{Y_{(1)},\dots,Y_{(n)}}(y_1,\dots,y_n)=n!\,f(y_1)f(y_2)\cdots f(y_n).
\]
Un par $(Y_{(i)},Y_{(j)})$, $i<j$, con $x<y$:
\[
f_{Y_{(i)},Y_{(j)}}(x,y)=\frac{n!}{(i-1)!\,(j-i-1)!\,(n-j)!}\,
f(x)f(y)\,[F(x)]^{i-1}\,[F(y)-F(x)]^{j-i-1}\,[1-F(y)]^{n-j}.
\]
\end{thmbox}

\begin{notebox}[frametitle={Casos que conviene recordar}]
\begin{itemize}
  \item Mínimo de exponenciales i.i.d.: $Y_{(1)}\sim\text{Exp}(\beta/n)$ (las
        tasas se suman).
  \item $Y_{(i)}$ de uniformes $U(0,1)$: $\text{Beta}(i,\,n-i+1)$.
\end{itemize}
\end{notebox}

\section{Distribuciones muestrales: $\chi^2$ y $t$ \textmd{(Clase 3)}}

Muestra $Y_1,\dots,Y_n$ i.i.d.\ de $N(\mu,\sigma^2)$;\quad
$\bar Y=\frac1n\sum Y_i$,\quad $S^2=\frac{1}{n-1}\sum(Y_i-\bar Y)^2$.

\begin{thmbox}[frametitle={Teorema 1: la media muestral}]
\[
\bar Y\sim N\!\left(\mu,\frac{\sigma^2}{n}\right)
\qquad\Longleftrightarrow\qquad
Z=\frac{\bar Y-\mu}{\sigma/\sqrt n}\sim N(0,1).
\]
Error estándar: $\sigma/\sqrt n$. (``Para reducir el error a la mitad, cuatro
veces más datos''.)
\end{thmbox}

\begin{defbox}[frametitle={Chi-cuadrado: $\chi^2_{(\nu)}=G(\nu/2,2)$}]
\[
\E(W)=\nu,\qquad \Var(W)=2\nu,\qquad M_W(t)=(1-2t)^{-\nu/2}.
\]
Aditiva: $\chi^2_{(\nu_1)}+\chi^2_{(\nu_2)}\sim\chi^2_{(\nu_1+\nu_2)}$
(independientes).
\end{defbox}

\begin{thmbox}[frametitle={Lema 1 y Teorema 2}]
\[
\sum_{i=1}^n\left(\frac{Y_i-\mu}{\sigma}\right)^2\sim\chi^2_{(n)};\qquad
\bar Y\ind S^2,\qquad
\frac{(n-1)S^2}{\sigma^2}=\frac{1}{\sigma^2}\sum_{i=1}^n(Y_i-\bar Y)^2\sim\chi^2_{(n-1)} .
\]
Se pierde un grado de libertad al reemplazar $\mu$ por $\bar Y$.
\end{thmbox}

\begin{defbox}[frametitle={t de Student}]
\[
\E(X)=0\ (\nu>1),\qquad \Var(X)=\frac{\nu}{\nu-2}\ (\nu>2);\qquad
t_{(\nu)}\to N(0,1)\ \text{si }\nu\to\infty .
\]
\end{defbox}

\begin{thmbox}[frametitle={Teorema 3: de dónde sale la $t$}]
\[
Z\sim N(0,1),\ W\sim\chi^2_{(\nu)},\ Z\ind W
\ \Longrightarrow\ T=\frac{Z}{\sqrt{W/\nu}}\sim t_{(\nu)} .
\]
Con $\sigma$ desconocida y $\nu=n-1$:\quad
$\dfrac{\bar Y-\mu}{S/\sqrt n}\sim t_{(n-1)}$ (el pivote para $\mu$ sin conocer
$\sigma$).
\end{thmbox}

\section{La distribución F \textmd{(Clase 4)}}

\begin{defbox}[frametitle={Definición (Fisher–Snedecor)}]
\[
f_X(x)=\frac{\left(\frac{\nu_1}{\nu_2}\right)^{\nu_1/2}}{\mathrm{B}(\nu_1/2,\nu_2/2)}\;
x^{\nu_1/2-1}\left(1+\frac{\nu_1}{\nu_2}x\right)^{-\frac{\nu_1+\nu_2}{2}},\qquad x>0 .
\]
$\nu_1$ = g.l.\ del \textbf{numerador}, $\nu_2$ = g.l.\ del \textbf{denominador};
el orden importa.
\end{defbox}

\begin{thmbox}[frametitle={Teorema 4}]
\[
W_1\sim\chi^2_{(\nu_1)}\ind W_2\sim\chi^2_{(\nu_2)}
\ \Longrightarrow\ F=\frac{W_1/\nu_1}{W_2/\nu_2}\sim F_{(\nu_1,\nu_2)} .
\]
\end{thmbox}

\begin{thmbox}[frametitle={Momentos}]
\[
\E(X)=\frac{\nu_2}{\nu_2-2}\ (\nu_2>2),\qquad
\Var(X)=\frac{2\nu_2^2(\nu_1+\nu_2-2)}{\nu_1(\nu_2-2)^2(\nu_2-4)}\ (\nu_2>4).
\]
\end{thmbox}

\begin{thmbox}[frametitle={Resultados a)–d)}]
\begin{itemize}
  \item \textbf{a)} $X\sim t_{(\nu)}\Rightarrow X^2\sim F_{(1,\nu)}$, y
        $F_{(1,\nu,\alpha)}=t_{(\nu),\alpha/2}^2$.
  \item \textbf{b)} $X\sim F_{(\nu_1,\nu_2)}\Rightarrow 1/X\sim F_{(\nu_2,\nu_1)}$.
  \item \textbf{c)} $F_{(\nu_1,\nu_2,1-\alpha)}=\dfrac{1}{F_{(\nu_2,\nu_1,\alpha)}}$.
  \item \textbf{d)} muestras normales independientes:
        $\dfrac{S_1^2/\sigma_1^2}{S_2^2/\sigma_2^2}\sim F_{(n_1-1,n_2-1)}$; si
        $\sigma_1^2=\sigma_2^2$, $S_1^2/S_2^2\sim F_{(n_1-1,n_2-1)}$.
\end{itemize}
\end{thmbox}

\section{Teorema del límite central \textmd{(Clase 5)}}

\begin{thmbox}[frametitle={Teorema 5 (TLC)}]
Sean $Y_1,Y_2,\dots$ i.i.d.\ con $\E[Y_i]=\mu$ y $\Var[Y_i]=\sigma^2<\infty$.
Con $S_n=\sum_{i=1}^n Y_i$ y
\[
W_n=\frac{\bar Y-\mu}{\sigma/\sqrt n}=\frac{S_n-n\mu}{\sigma\sqrt n}\ \dto\ N(0,1):
\qquad
\lim_{n\to\infty}\Prob(W_n\le w)=\Phi(w)\quad\forall w\in\R .
\]
\end{thmbox}

\begin{notebox}[frametitle={Formas equivalentes (en la práctica, para $n$ grande)}]
\[
Y_i\ \text{i.i.d.}\ \Longrightarrow\
\bar Y\ \adist\ N\!\left(\mu,\frac{\sigma^2}{n}\right),\qquad
S_n=\sum_{i=1}^n Y_i\ \adist\ N\!\left(n\mu,\ n\sigma^2\right).
\]
Calcula $\mu$ y $\sigma^2$ de \textbf{una sola} observación (Formulario 2). Caso
Bernoulli: $\text{Bin}(n,p)\adist N\big(np,\,np(1-p)\big)$ si $np\ge5$ y
$n(1-p)\ge5$.
\end{notebox}

\begin{defbox}[frametitle={Corrección por continuidad ($X$ entera)}]
Escribe el evento con $\le$ o $\ge$ y extremos enteros, luego ensancha medio
punto hacia afuera.

\smallskip
\begingroup\footnotesize
\setlength{\tabcolsep}{6pt}
\noindent
\begin{tabular}{@{}l p{11cm}@{}}
\toprule
\textbf{Evento} & \textbf{Aproximación}\\
\midrule
$X\le k$ & $\Phi\!\big(\frac{k+0.5-\mu_X}{\sigma_X}\big)$\\
$X<k$ & $\Phi\!\big(\frac{k-0.5-\mu_X}{\sigma_X}\big)$\\
$X\ge k$ & $1-\Phi\!\big(\frac{k-0.5-\mu_X}{\sigma_X}\big)$\\
$X>k$ & $1-\Phi\!\big(\frac{k+0.5-\mu_X}{\sigma_X}\big)$\\
$X=k$ & $\Phi\!\big(\frac{k+0.5-\mu_X}{\sigma_X}\big)-\Phi\!\big(\frac{k-0.5-\mu_X}{\sigma_X}\big)$\\
$a\le X\le b$ & $\Phi\!\big(\frac{b+0.5-\mu_X}{\sigma_X}\big)-\Phi\!\big(\frac{a-0.5-\mu_X}{\sigma_X}\big)$\\
\bottomrule
\end{tabular}
\endgroup

No se usa si la variable es continua; con rejilla de paso $h$, la corrección es
$\pm h/2$.
\end{defbox}

\begin{warnbox}[frametitle={Sin varianza finita no hay TLC}]
$Y_i\iid\text{Cauchy}(0,1)$ (fdp $\frac{1}{\pi(1+y^2)}$, sin media ni varianza)
$\Rightarrow \bar Y\sim\text{Cauchy}(0,1)$ para todo $n$: ni se concentra ni se
vuelve normal.
\end{warnbox}

\section{Estimación puntual, sesgo y error cuadrático medio \textmd{(Clase 6)}}

\begin{defbox}[frametitle={Estimador puntual frente a estimación puntual}]
Sea $X_1,\dots,X_n$ una muestra aleatoria de una población cuya distribución depende de un parámetro desconocido $\theta\in\Omega$.
\begin{itemize}
  \item \textbf{Estimador puntual:} estadístico $\hat\theta = T(X_1,\dots,X_n)$. Es una \textbf{variable aleatoria} (función de la muestra) que no depende de parámetros desconocidos; posee distribución muestral, esperanza y varianza.
  \item \textbf{Estimación puntual:} número real fijado $\hat\theta(x_1,\dots,x_n) = T(x_1,\dots,x_n)$, correspondiente al valor observado del estimador al evaluarlo en una muestra concreta.
\end{itemize}
\end{defbox}

\begin{defbox}[frametitle={Insesgamiento y sesgo}]
Un estimador $\hat\theta$ es \textbf{insesgado} para $\theta$ si su valor esperado coincide con el parámetro:
\[
\E[\hat\theta] = \theta \quad\forall\theta\in\Omega
\qquad\Longleftrightarrow\qquad
B(\hat\theta) = \E[\hat\theta] - \theta = 0.
\]
Si $\E[\hat\theta]\ne\theta$, el estimador es \textbf{sesgado} con sesgo $B(\hat\theta) = \E[\hat\theta] - \theta$.
\end{defbox}

\begin{thmbox}[frametitle={Descomposición del Error Cuadrático Medio (MSE)}]
El \textbf{error cuadrático medio} ($\text{MSE}_\theta(\hat\theta)$) cuantifica la discrepancia cuadrática global entre el estimador y el parámetro:
\[
\text{MSE}_\theta(\hat\theta) = \E\big[(\hat\theta - \theta)^2\big] = \Var(\hat\theta) + \big[B(\hat\theta)\big]^2 .
\]
En palabras: $\text{MSE} = \text{Varianza} + (\text{Sesgo})^2$.
Si $\hat\theta$ es \textbf{insesgado} ($B(\hat\theta)=0$):
\[
\text{MSE}_\theta(\hat\theta) = \Var(\hat\theta).
\]
\end{thmbox}

\begin{notebox}[frametitle={Estimadores fundamentales en población normal $N(\mu,\sigma^2)$}]
Muestra $X_1,\dots,X_n \iid N(\mu,\sigma^2)$ con $n \ge 2$:

\smallskip
\begingroup\footnotesize
\setlength{\tabcolsep}{4pt}
\noindent
\begin{tabular}{@{}p{3.6cm} p{1.4cm} p{3.1cm} p{3.4cm} p{3.4cm}@{}}
\toprule
\textbf{Estimador} & $\mathbf{\E}$ & \textbf{Sesgo} $B$ & $\mathbf{\Var}$ & $\mathbf{\text{MSE}}$\\
\midrule
\rr Media: $\bar X = \frac{1}{n}\sum_{i=1}^n X_i$ & $\mu$ & $0$ (insesgado) & $\dfrac{\sigma^2}{n}$ & $\dfrac{\sigma^2}{n}$\\[6pt]
\rr Cuasivarianza: $S^2 = \frac{1}{n-1}\sum_{i=1}^n (X_i-\bar X)^2$ & $\sigma^2$ & $0$ (insesgado) & $\dfrac{2\sigma^4}{n-1}$ & $\dfrac{2\sigma^4}{n-1}$\\[6pt]
\rr Varianza muestral: $S_n^2 = \frac{1}{n}\sum_{i=1}^n (X_i-\bar X)^2$ & $\dfrac{n-1}{n}\sigma^2$ & $-\dfrac{\sigma^2}{n} < 0$ (a la baja) & $\dfrac{2(n-1)\sigma^4}{n^2}$ & $\dfrac{2n-1}{n^2}\sigma^4$\\[6pt]
\rr Con dos datos: $\frac{(X_1-X_2)^2}{2}$ & $\sigma^2$ & $0$ (insesgado) & $2\sigma^4$ & $2\sigma^4$\\
\bottomrule
\end{tabular}
\endgroup

\smallskip
\textbf{Corrección de Bessel:} $S^2 = \frac{n}{n-1}S_n^2$. Como $\sum(X_i-\bar X)^2$ minimiza la suma de cuadrados de desviaciones respecto a cualquier punto, $S_n^2$ subestima sistemáticamente a $\sigma^2$; multiplicar por $\frac{n}{n-1}$ elimina exactamente dicho sesgo negativo.
\end{notebox}

\section{Insesgamiento asintótico, cota del error y tamaño de muestra \textmd{(Clase 7)}}

\begin{defbox}[frametitle={Estimador asintóticamente insesgado}]
Una sucesión de estimadores $\{\hat\theta_n\}$ es \textbf{asintóticamente insesgada} si su sesgo tiende a cero cuando el tamaño de muestra $n$ tiende a infinito:
\[
\lim_{n\to\infty} B(\hat\theta_n) = 0
\qquad\Longleftrightarrow\qquad
\lim_{n\to\infty} \E[\hat\theta_n] = \theta .
\]
\begin{itemize}
  \item \textbf{Ejemplo asintóticamente insesgado:} $S_n^2$ tiene sesgo $B(S_n^2) = -\frac{\sigma^2}{n} \to 0$ cuando $n\to\infty$ (sí lo es).
  \item \textbf{Ejemplo no asintóticamente insesgado:} para estimar $\theta = \lambda^2$ con $X_i\iid\text{Poisson}(\lambda)$, el estimador $T_n = (\bar X)^2 - \bar X$ tiene $\E[T_n] = \lambda^2 - \frac{n-1}{n}\lambda \to \lambda^2 - \lambda$, por lo que $B(T_n) \to -\lambda \ne 0$ (no lo es). Estimador insesgado corregido: $\hat\theta_n = (\bar X)^2 - \frac{\bar X}{n}$.
\end{itemize}
\end{defbox}

\begin{defbox}[frametitle={Error de estimación y cota $b$}]
El \textbf{error de estimación} es la distancia absoluta $\varepsilon = |\hat\theta - \theta|$. Se exige que $\Prob(\varepsilon < b) = 1-\alpha$ para una tolerancia admisible $b > 0$.
\begin{itemize}
  \item \textbf{Cota empírica (regla de dos desvíos, confianza $\approx 95\%$):} por el TLC y la regla empírica,
  \[
  b = 2\sigma_{\hat\theta} = 2\sqrt{\Var(\hat\theta)}.
  \]
  \item \textbf{Forma exacta con cuantil normal estándar:} $b = z_{\alpha/2}\,\sigma_{\hat\theta}$ (para $95\%$, $z_{0.025} \approx 1.96$).
\end{itemize}
\end{defbox}

\begingroup\footnotesize
\setlength{\tabcolsep}{5pt}
\noindent
\begin{tabular}{@{}p{3.5cm} p{2.4cm} p{4.6cm} p{5.5cm}@{}}
\toprule
\textbf{Parámetro objetivo} $\theta$ & \textbf{Estimador} $\hat\theta$ & \textbf{Error estándar} $\sigma_{\hat\theta}$ & \textbf{Propiedad / Cota máxima}\\
\midrule
\rr Media $\mu$ & $\bar Y$ & $\sigma_{\bar Y} = \dfrac{\sigma}{\sqrt n}$ & \rr Insesgado: $\E[\bar Y]=\mu$\\[7pt]
\rr Proporción $p$ & $\hat p = \dfrac{Y}{n}$ & $\sigma_{\hat p} = \sqrt{\dfrac{p(1-p)}{n}}$ & \rr $\sigma_{\hat p} \le \dfrac{1}{2\sqrt n}$ (máximo en $p = 0.5$)\\[9pt]
\rr Dif.\ de medias $\mu_1 - \mu_2$ & $\bar Y_1 - \bar Y_2$ & $\sigma_{\bar Y_1-\bar Y_2} = \sqrt{\dfrac{\sigma_1^2}{n_1} + \dfrac{\sigma_2^2}{n_2}}$ & \rr Muestras independientes\\[9pt]
\rr Dif.\ proporciones $p_1 - p_2$ & $\hat p_1 - \hat p_2$ & $\sigma_{\hat p_1-\hat p_2} = \sqrt{\dfrac{p_1(1-p_1)}{n_1} + \dfrac{p_2(1-p_2)}{n_2}}$ & \rr Muestras independientes\\
\bottomrule
\end{tabular}
\endgroup

\begin{thmbox}[frametitle={Tamaño de muestra requerido para cota de error $b$}]
Despejando $n$ de la cota empírica $b = 2\sigma_{\hat\theta}$ ($\approx 95\%$):
\begin{itemize}
  \item \textbf{Para la media $\mu$:}
  \[
  2\frac{\sigma}{\sqrt n} = b \quad\Longrightarrow\quad n = \left(\frac{2\sigma}{b}\right)^2 .
  \]
  \textit{Regla de cuadruplicación:} reducir el margen $b$ a la mitad exige \textbf{cuatro veces} más datos ($n \propto 1/b^2$).
  \item \textbf{Para la proporción $p$:}
  \[
  2\sqrt{\frac{p(1-p)}{n}} = b \quad\Longrightarrow\quad n = \frac{4p(1-p)}{b^2} \le \frac{1}{b^2} .
  \]
  La cota \textbf{conservadora} (peor caso) se alcanza en $p = 0.5$, garantizando que el error no supere $b$ sin requerir conocimiento previo de $p$.
\end{itemize}
\end{thmbox}

\section{Métodos de estimación: Momentos y Máxima Verosimilitud \textmd{(Clase 8)}}

\begin{defbox}[frametitle={Método de los Momentos (MM)}]
Sea $X_1,\dots,X_n$ una muestra aleatoria $\iid$ con fdp o fmp $f(x;\theta_1,\dots,\theta_K)$.
\begin{itemize}
  \item \textbf{Momentos poblacionales (teóricos):} $\mu_k(\theta) = \E[X^k]$, para $k \ge 1$.
  \item \textbf{Momentos muestrales (empíricos):} $m_k = \frac{1}{n}\sum_{i=1}^n X_i^k$, con $m_1 = \bar X$.
  \item \textbf{Sistema de ecuaciones MM:} se igualan los primeros $K$ momentos muestrales a los momentos teóricos:
  \[
  m_k = \mu_k(\hat\theta_1,\dots,\hat\theta_K), \qquad k=1,\dots,K.
  \]
  El estimador MM $(\hat\theta_1^{\text{MM}},\dots,\hat\theta_K^{\text{MM}})$ es la solución algebraica del sistema.
  \item \textbf{Atajo con momentos centrales y varianza:}
  \[
  \mu_1 = \mu, \qquad \sigma^2 = \mu_2 - \mu_1^2 \quad\Longrightarrow\quad \hat\sigma^2_{\text{MM}} = m_2 - m_1^2 = \frac{1}{n}\sum_{i=1}^n (X_i - \bar X)^2 = S_n^2 .
  \]
  \item \textbf{Justificación asintótica:} por la LFGN, $m_k \dcs \mu_k$; proporciona estimadores consistentes bajo condiciones de regularidad.
\end{itemize}
\end{defbox}

\begin{defbox}[frametitle={Función de Verosimilitud y Log-Verosimilitud}]
Sea $\mathbf{x} = (x_1,\dots,x_n)$ una muestra observada $\iid$ con función de densidad o masa $f(x;\theta)$, $\theta\in\Theta$.
\begin{itemize}
  \item \textbf{Función de Verosimilitud:} evalúa la verosimilitud de los parámetros dados los datos observados fijos:
  \[
  L(\theta) = L(\theta;\mathbf{x}) = \prod_{i=1}^n f(x_i;\theta).
  \]
  \item \textbf{Función de Log-Verosimilitud:} transforma productos en sumas y preserva el punto donde se alcanza el máximo:
  \[
  \ell(\theta) = \ln L(\theta) = \sum_{i=1}^n \ln f(x_i;\theta).
  \]
\end{itemize}
\end{defbox}

\begin{thmbox}[frametitle={Estimador de Máxima Verosimilitud (EMV)}]
El \textbf{Estimador de Máxima Verosimilitud} (EMV o MLE) $\hat\theta_{\text{MV}}$ es el argumento que maximiza la verosimilitud:
\[
\hat\theta_{\text{MV}} = \arg\max_{\theta\in\Theta} L(\theta) = \arg\max_{\theta\in\Theta} \ell(\theta).
\]
\textbf{Condiciones de optimización en el interior de $\Theta$ (casos regulares):}
\begin{enumerate}
  \item \textbf{Condición de primer orden (Ecuación de verosimilitud):}
  \[
  \frac{\partial \ell(\theta)}{\partial \theta} = 0 \qquad \big(\nabla_\theta \ell(\boldsymbol{\theta}) = \mathbf{0} \text{ para } K \text{ parámetros}\big).
  \]
  \item \textbf{Condición de segundo orden (Concavidad / Máximo local):}
  \[
  \left.\frac{\partial^2 \ell(\theta)}{\partial \theta^2}\right|_{\theta=\hat\theta} < 0 \qquad \big(\mathbf{H}_\ell(\hat{\boldsymbol{\theta}}) \text{ matriz hessiana definida negativa}\big).
  \]
\end{enumerate}
\end{thmbox}

\begin{warnbox}[frametitle={Soporte dependiente del parámetro (Fallo del cálculo diferencial)}]
Si el soporte de la población depende de $\theta$, la verosimilitud no es derivable en la frontera y \textbf{la derivada $\ell'(\theta)=0$ falla}:
\begin{itemize}
  \item \textbf{Ejemplo canónico $X_i \iid U(0,\theta)$:} densidad $f(x;\theta) = \frac{1}{\theta}\,\mathbb{I}_{(0 \le x \le \theta)}$. La verosimilitud conjunta impone que $\theta \ge x_i$ para todo $i$, es decir, $\theta \ge X_{(n)} = \max(X_1,\dots,X_n)$:
  \[
  L(\theta) = \frac{1}{\theta^n}\,\mathbb{I}_{(\theta \ge X_{(n)})} \mathbb{I}_{(X_{(1)} \ge 0)}.
  \]
  Como $\theta^{-n}$ es estrictamente decreciente para $\theta \ge X_{(n)}$, el valor admisible que maximiza $L(\theta)$ es la cota inferior:
  \[
  \hat\theta_{\text{MV}} = X_{(n)}.
  \]
  \item Regla práctica: límite superior dependiente de $\theta \implies \hat\theta_{\text{MV}} = X_{(n)}$; límite inferior dependiente de $\theta \implies \hat\theta_{\text{MV}} = X_{(1)}$.
\end{itemize}
\end{warnbox}

\section{Estimadores eficientes y Cota de Cramér-Rao \textmd{(Clase 9)}}

\begin{defbox}[frametitle={Función Score de Fisher}]
Sea $X$ una v.a.\ con fdp o fmp $f(x;\theta)$, $\theta\in\Theta\subseteq\R$.
\begin{itemize}
  \item \textbf{Función Score individual:} derivada del logaritmo de la densidad de una observación:
  \[
  S(\theta) = S(X;\theta) = \frac{\partial \ln f(X;\theta)}{\partial \theta} .
  \]
  \item \textbf{Función Score muestral:} para una muestra $\iid$ $X_1,\dots,X_n$:
  \[
  S_n(\theta) = \frac{\partial \ell(\theta)}{\partial \theta} = \sum_{i=1}^n S(X_i;\theta).
  \]
  \item \textbf{Esperanza nula de la Score:} bajo condiciones de regularidad (intercambio integral-derivada):
  \[
  \E[S(\theta)] = 0 \qquad\text{y}\qquad \E[S_n(\theta)] = 0 .
  \]
\end{itemize}
\end{defbox}

\begin{defbox}[frametitle={Información de Fisher}]
La \textbf{Información de Fisher} mide la curvatura esperada de la log-verosimilitud (cantidad de información que la muestra aporta sobre $\theta$):
\begin{itemize}
  \item \textbf{Información de Fisher de una sola observación:}
  \[
  I_F(\theta) = \Var\big(S(\theta)\big) = \E\big[S(\theta)^2\big] = -\E\left[\frac{\partial^2 \ln f(X;\theta)}{\partial \theta^2}\right].
  \]
  La forma de la segunda derivada ($I_F(\theta) = -\E[\ell'']$) es la más eficiente para el cálculo analítico.
  \item \textbf{Información muestral (Aditividad para muestra $\iid$):}
  \[
  I_n(\theta) = \Var\big(S_n(\theta)\big) = n\,I_F(\theta).
  \]
\end{itemize}
\end{defbox}

\begin{thmbox}[frametitle={Cota Inferior de Cramér-Rao (CRLB)}]
Bajo condiciones de regularidad (soporte independiente de $\theta$, existencia de derivadas segundas e intercambio con la integral):
\begin{enumerate}
  \item \textbf{Estimador insesgado ($\E[\hat\theta]=\theta$):}
  \[
  \Var(\hat\theta) \ge \frac{1}{I_n(\theta)} = \frac{1}{n\,I_F(\theta)} = \text{CRLB}(\theta).
  \]
  \item \textbf{Estimador con sesgo $B(\theta) = \E[\hat\theta] - \theta$ (es decir $\E[\hat\theta] = \theta + B(\theta)$):}
  \[
  \Var(\hat\theta) \ge \frac{[1 + B'(\theta)]^2}{n\,I_F(\theta)}.
  \]
\end{enumerate}
\end{thmbox}

\begin{thmbox}[frametitle={Eficiencia absoluta y estimador UMVUE}]
Sea $\hat\theta$ un estimador insesgado de $\theta$:
\begin{itemize}
  \item \textbf{Eficiencia absoluta:}
  \[
  \text{eff}(\hat\theta) = \frac{\text{CRLB}(\theta)}{\Var(\hat\theta)} = \frac{1/\big(n\,I_F(\theta)\big)}{\Var(\hat\theta)}, \qquad 0 \le \text{eff}(\hat\theta) \le 1.
  \]
  \item \textbf{Estimador eficiente y UMVUE:} si $\text{eff}(\hat\theta) = 1$, $\hat\theta$ es \textbf{eficiente} y constituye el Estimador Insesgado de Varianza Mínima Uniforme (\textbf{UMVUE}).
  \item \textbf{Condición de logro de la cota (Igualdad en Cauchy-Schwarz):} $\Var(\hat\theta) = \text{CRLB}$ si y solo si:
  \[
  \frac{\partial \ell(\theta)}{\partial \theta} = k(\theta, n)\,\big[\hat\theta(\mathbf{x}) - \theta\big].
  \]
\end{itemize}
\end{thmbox}

\begin{warnbox}[frametitle={Condiciones de regularidad de Cramér-Rao y excepciones}]
La Cota de Cramér-Rao exige que el soporte de la distribución \textbf{no dependa} del parámetro $\theta$. En modelos no regulares (como $U(0,\theta)$ con estimador insesgado $\hat\theta = \frac{n+1}{n}X_{(n)}$), la varianza decrece a orden $\mathcal{O}(1/n^2)$, superando el límite $\mathcal{O}(1/n)$ de Cramér-Rao porque no se cumplen las condiciones de regularidad.
\end{warnbox}

\section{Invarianza, Propiedades Asintóticas del EMV y Consistencia \textmd{(Clase 10)}}

\begin{thmbox}[frametitle={Principio de Invarianza de Zehna (EMV de funciones)}]
Sea $\hat\theta_{\text{MV}}$ el estimador de máxima verosimilitud de $\theta\in\Theta$. Para cualquier función $g: \Theta \to \Omega$ (inyectiva o no inyectiva), el EMV del parámetro transformado $\tau = g(\theta)$ es:
\[
\widehat{g(\theta)}_{\text{MV}} = g\big(\hat\theta_{\text{MV}}\big).
\]
\textbf{Advertencia:} la invarianza preserva la verosimilitud, \textbf{no el insesgamiento}. Si $g$ no es lineal, por la desigualdad de Jensen $\E[g(\hat\theta)] \ne g(\theta)$, lo que suele introducir sesgo en muestras finitas (que se desvanece asintóticamente).
\end{thmbox}

\begin{thmbox}[frametitle={Propiedades Asintóticas del EMV y Normalidad Límite}]
Bajo condiciones de regularidad, cuando el tamaño de muestra $n \to \infty$, el EMV $\hat\theta_{\text{MV}}$ satisface:
\begin{enumerate}
  \item \textbf{Consistencia:} $\hat\theta_n \pto \theta$ (y fuertemente $\hat\theta_n \dcs \theta$).
  \item \textbf{Normalidad asintótica y eficiencia asintótica ($100\%$):}
  \[
  \sqrt{n}\big(\hat\theta_{\text{MV}} - \theta\big) \dto N\left(0, \frac{1}{I_F(\theta)}\right)
  \qquad\Longleftrightarrow\qquad
  \hat\theta_{\text{MV}} \adist N\left(\theta, \frac{1}{n\,I_F(\theta)}\right) = N\big(\theta, \text{CRLB}(\theta)\big).
  \]
  El EMV alcanza asintóticamente la Cota de Cramér-Rao con eficiencia del $100\%$.
\end{enumerate}
\end{thmbox}

\begin{thmbox}[frametitle={Método Delta (Aproximación de transformaciones 1D)}]
Sea $\{T_n\}$ una sucesión de estimadores con $\sqrt{n}(T_n - \theta) \dto N(0, \sigma^2)$, y sea $g$ derivable con $g'(\theta) \ne 0$. Entonces:
\[
\sqrt{n}\big(g(T_n) - g(\theta)\big) \dto N\Big(0,\, \sigma^2 \big[g'(\theta)]^2\Big).
\]
Para el estimador de máxima verosimilitud ($\sigma^2 = 1/I_F(\theta)$):
\[
\sqrt{n}\Big(g(\hat\theta_{\text{MV}}) - g(\theta)\Big) \dto N\left(0,\, \frac{[g'(\theta)]^2}{I_F(\theta)}\right)
\qquad\Longleftrightarrow\qquad
\Var\big(g(\hat\theta_{\text{MV}})\big) \approx \frac{[g'(\theta)]^2}{n\,I_F(\theta)}.
\]
\end{thmbox}

\begin{defbox}[frametitle={Consistencia y Convergencia en Probabilidad}]
Una sucesión de estimadores $\{\hat\theta_n\}$ es \textbf{consistente} para $\theta$ ($\hat\theta_n \pto \theta$) si para todo $\varepsilon > 0$:
\[
\lim_{n\to\infty} \Prob\big(|\hat\theta_n - \theta| \ge \varepsilon\big) = 0
\qquad\Longleftrightarrow\qquad
\lim_{n\to\infty} \Prob\big(|\hat\theta_n - \theta| < \varepsilon\big) = 1.
\]
\textbf{Criterio suficiente del Error Cuadrático Medio (MSE):}
\[
\lim_{n\to\infty} \text{MSE}(\hat\theta_n) = 0 \quad\Longleftrightarrow\quad \left(\lim_{n\to\infty}\Var(\hat\theta_n) = 0 \ \text{ y }\ \lim_{n\to\infty} B(\hat\theta_n) = 0\right) \quad\Longrightarrow\quad \hat\theta_n \pto \theta .
\]
\end{defbox}

\begin{notebox}[frametitle={Desigualdades clásicas fundamentales}]
\begin{itemize}
  \item \textbf{Desigualdad de Markov Generalizada (DMG):} para cualquier v.a.\ $X$, constante $t > 0$ y $\lambda > 0$:
  \[
  \Prob(|X| \ge \lambda) \le \frac{\E[|X|^t]}{\lambda^t}.
  \qquad \text{Markov clásica ($X\ge 0, t=1$): } \Prob(X \ge a) \le \frac{\E[X]}{a}.
  \]
  \item \textbf{Desigualdad de Chebyshev:} para variable con media $\mu$ y varianza finita $\sigma^2$ (DMG con $t=2$):
  \[
  \Prob(|X - \mu| \ge \varepsilon) \le \frac{\Var(X)}{\varepsilon^2}, \qquad \Prob(|X - \mu| \ge k\sigma) \le \frac{1}{k^2}.
  \]
  \item \textbf{Desigualdad de Jensen:} si $g$ es una función convexa ($g''(u) \ge 0$) y $\E[|X|] < \infty$:
  \[
  \E[g(X)] \ge g\big(\E[X]\big). \qquad \big(\text{Ejemplo: } \E[X^2] \ge (\E[X])^2 \implies \Var(X) \ge 0\big).
  \]
  \item \textbf{Desigualdad de Cauchy-Schwarz:} si $\E[X^2] < \infty$ y $\E[Y^2] < \infty$:
  \[
  \big|\E[XY]\big| \le \sqrt{\E[X^2]\,\E[Y^2]}. \qquad \big(\text{Centradas: } |\Cov(X,Y)| \le \sigma_X \sigma_Y \iff |\rho| \le 1\big).
  \]
\end{itemize}
\end{notebox}

\begin{warnbox}[frametitle={Robustez de estimadores y Punto de Ruptura (Breakdown Point)}]
Un estimador es \textbf{robusto} si es insensible a violaciones de supuestos del modelo y a observaciones atípicas extremas (outliers).
\begin{itemize}
  \item \textbf{Punto de ruptura ($\varepsilon^*$):} proporción mínima de datos contaminados arbitrariamente grandes que pueden forzar al estimador al infinito ($\pm\infty$).
  \item \textbf{Media muestral $\bar X$:} $\varepsilon^* = \frac{1}{n} \xrightarrow{n\to\infty} 0\%$ (\textbf{no robusta}; un solo valor atípico altera la media sin límite).
  \item \textbf{Mediana muestral $\tilde X$:} $\varepsilon^* \approx 50\%$ (\textbf{máxima robustez}; tolera casi el $50\%$ de datos aberrantes).
\end{itemize}
\end{warnbox}

\section{Modos de convergencia, Borel--Cantelli y mapeo continuo \textmd{(Clase 11)}}

\begin{defbox}[frametitle={Los tres modos de convergencia estocástica}]
Sean $X_1, X_2, \dots$ y $X$ variables aleatorias sobre $(\Omega, \mathcal{F}, \Prob)$ (en distribución sólo importan las marginales $F_{X_n}$ y $F_X$):
\begin{itemize}
  \item \textbf{Casi segura ($X_n \dcs X$):} el conjunto de trayectorias donde converge puntualmente tiene probabilidad $1$:
  \[
  \Prob\big(\{\omega \in \Omega : \lim_{n\to\infty} X_n(\omega) = X(\omega)\}\big) = 1
  \;\Longleftrightarrow\;
  \Prob\big(|X_n - X| > \varepsilon \text{ i.v.}\big) = 0 \quad \forall\,\varepsilon > 0,
  \]
  donde ``i.v.'' (infinitas veces) es el límite superior conjuntista $\{A_n \text{ i.v.}\} = \limsup_{n\to\infty} A_n = \bigcap_{m=1}^\infty \bigcup_{n=m}^\infty A_n$.
  \item \textbf{En probabilidad ($X_n \pto X$):} para todo $\varepsilon > 0$,
  \[
  \lim_{n\to\infty} \Prob\big(|X_n - X| > \varepsilon\big) = 0
  \;\Longleftrightarrow\;
  \lim_{n\to\infty} \Prob\big(|X_n - X| \le \varepsilon\big) = 1.
  \]
  \item \textbf{En distribución ($X_n \dto X$):} convergencia puntual de las acumuladas en los puntos de continuidad del límite:
  \[
  \lim_{n\to\infty} F_{X_n}(x) = F_X(x) \quad \text{en todo } x \in \R \text{ donde } F_X \text{ es continua}.
  \]
\end{itemize}
\end{defbox}

\begin{thmbox}[frametitle={Jerarquía, unicidad del límite y equivalencia hacia una constante}]
\begin{itemize}
  \item \textbf{Jerarquía estricta:} $X_n \dcs X \implies X_n \pto X \implies X_n \dto X$. Ninguna implicación recíproca vale en general sin hipótesis adicionales.
  \item \textbf{Unicidad del límite:} si $X_n \to X$ y $X_n \to Y$ en c.s.\ o en $p$, entonces $\Prob(X = Y) = 1$; si convergen en $d$, entonces $F_X(x) = F_Y(x)$ para todo $x \in \R$.
  \item \textbf{Equivalencia hacia una constante $c \in \R$:} para una variable degenerada en $c$ ($F_c(x) = \mathbf{1}_{\{x \ge c\}}$, continua salvo en $x = c$):
  \[
  X_n \dto c \;\Longleftrightarrow\; X_n \pto c.
  \]
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Lemas de Borel--Cantelli y criterio suficiente c.s.}]
\begin{itemize}
  \item \textbf{1.\textsuperscript{er} Lema de Borel--Cantelli (no requiere independencia):} si la serie de probabilidades converge, entonces con probabilidad $1$ ocurre a lo sumo un número finito de eventos $A_n$:
  \[
  \sum_{n=1}^\infty \Prob(A_n) < \infty \implies \Prob\big(\limsup_{n\to\infty} A_n\big) = \Prob(A_n \text{ i.v.}) = 0.
  \]
  \item \textbf{Criterio suficiente para convergencia c.s.:} tomando $A_n = \{|X_n - X| > \varepsilon\}$, si para todo $\varepsilon > 0$ se cumple
  \[
  \sum_{n=1}^\infty \Prob\big(|X_n - X| > \varepsilon\big) < \infty \implies X_n \dcs X \quad (\text{y por tanto } X_n \pto X).
  \]
  \item \textbf{2.\textsuperscript{o} Lema de Borel--Cantelli (exige eventos independientes):} si $\{A_n\}_{n\ge 1}$ son \textbf{independientes} y la serie diverge, entonces ocurren infinitas veces con probabilidad $1$:
  \[
  \sum_{n=1}^\infty \Prob(A_n) = \infty \text{ con } \{A_n\} \text{ independientes} \implies \Prob\big(\limsup_{n\to\infty} A_n\big) = \Prob(A_n \text{ i.v.}) = 1.
  \]
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Teorema del mapeo continuo y aritmética de límites}]
\begin{itemize}
  \item \textbf{Mapeo continuo:} si $g: \R \to \R$ es continua en el soporte de $X$ (o su conjunto de discontinuidades tiene probabilidad $0$ bajo $X$), entonces $X_n \to X \implies g(X_n) \to g(X)$ en el \textbf{mismo modo} ($\dcs$, $\pto$ o $\dto$). Asimismo, si $c_n \to c \in \R$ (determinista), $c_n X_n \to c X$ en ese mismo modo.
  \item \textbf{Aritmética en c.s.\ y en probabilidad:} si $X_n \pto a$ y $Y_n \pto b$ (o $X_n \to X$, $Y_n \to Y$ en $\dcs$ o $\pto$), entonces $X_n \pm Y_n \to X \pm Y$, $X_n Y_n \to XY$ y, si $\Prob(Y = 0) = 0$ (o $b \ne 0$), $X_n / Y_n \to X / Y$.
\end{itemize}
\end{thmbox}

\begin{warnbox}[frametitle={¡Cuidado con la suma en distribución!}]
Que $X_n \dto X$ y $Y_n \dto Y$ \textbf{no implica} en general que $X_n + Y_n \dto X + Y$ (la convergencia débil ignora la estructura conjunta), salvo bajo independencia ($X_n \ind Y_n$ y $X \ind Y$) o cuando una sucesión converge a una constante (Slutsky).
\end{warnbox}

\begin{notebox}[frametitle={Consistencia del EMV, velocidades de convergencia y máximo en $U(0,1)$}]
\begin{itemize}
  \item \textbf{Criterio ECM (MSE):} $\lim_{n\to\infty} \Var(\hat\theta_n) = 0$ y $\lim_{n\to\infty} B(\hat\theta_n) = 0 \iff \lim_{n\to\infty} \text{MSE}_\theta(\hat\theta_n) = 0 \implies \hat\theta_n \pto \theta$. Bajo condiciones de regularidad, $\hat\theta_{\text{MV}} \pto \theta$ con tasa $\text{MSE} = \mathcal{O}(1/n)$.
  \item \textbf{Modelos no regulares ($U(-\theta, \theta)$):} cuando el soporte depende de $\theta$, $\hat\theta_{\text{MV}} = M_n = \max_{1\le i\le n}|X_i|$ cumple $M_n/\theta \sim \text{Beta}(n,1)$ y $\text{MSE}_\theta(M_n) = \dfrac{2\theta^2}{(n+1)(n+2)} = \mathcal{O}(1/n^2)$, mucho más rápido que el estimador de momentos $\hat\theta_{\text{MM}} = \sqrt{\frac{3}{n}\sum_{i=1}^n X_i^2}$ cuya tasa es $\text{MSE}_\theta(\hat\theta_{\text{MM}}) \approx \dfrac{\theta^2}{5n} = \mathcal{O}(1/n)$.
  \item \textbf{Límite del máximo en $U(0,1)$:} para $X_1,\dots,X_n \iid U(0,1)$, se cumple $X_{(n)} \pto 1$ y en escala $n$:
  \begin{gather*}
  n(1 - X_{(n)}) \dto \text{Exp}(1) \quad (\text{escala }\beta = 1),\\
  \text{y en general } n\,\text{Beta}(\alpha, n) \dto \text{Gamma}(\alpha, 1),\qquad \frac{n(\theta - M_n)}{\theta} \dto \text{Exp}(1).
  \end{gather*}
\end{itemize}
\end{notebox}

\section{Teorema de la Continuidad, Scheffé, Slutsky y Método Delta ampliado \textmd{(Clase 12)}}

\begin{thmbox}[frametitle={Teorema de la Continuidad (FGM / Lévy) y lema exponencial}]
Si las FGM $M_{X_n}(t)$ existen en $|t| < b$ ($b > 0$) y convergen puntualmente a la FGM de una variable aleatoria $X$:
\[
\lim_{n\to\infty} M_{X_n}(t) = M_X(t) \quad \text{para } |t| < h \;\;(0 < h \le b) \implies X_n \dto X.
\]
\textbf{Lema exponencial útil:} si $c_n \to c \in \R$ cuando $n \to \infty$, entonces $\displaystyle\lim_{n\to\infty}\left(1 + \frac{c_n}{n}\right)^n = e^c$.
\end{thmbox}

\begin{thmbox}[frametitle={Teorema de Scheffé, criterio discreto y límite de Poisson}]
\begin{itemize}
  \item \textbf{Discretas:} si $X_n, X$ toman valores en un conjunto numerable fijo (p.\ ej.\ $\mathbb{N}$) y $\lim_{n\to\infty} p_{X_n}(k) = p_X(k)$ para todo $k$, entonces $X_n \dto X$ (en $\mathbb{N}$, es condición necesaria y suficiente).
  \item \textbf{Teorema de Scheffé (continuas):} si $\lim_{n\to\infty} f_{X_n}(x) = f_X(x)$ para casi todo $x \in \R$, entonces $X_n \dto X$.
  \item \textbf{Límite de Poisson:} si $X_n \sim \text{Bin}(n, p_n)$ y $\lim_{n\to\infty} n p_n = \lambda > 0$, entonces $X_n \dto \text{Pois}(\lambda)$.
\end{itemize}
\end{thmbox}

\begin{warnbox}[frametitle={¡Cuidado! El recíproco del Teorema de Scheffé es falso}]
$X_n \dto X$ \textbf{no} implica $f_{X_n}(x) \to f_X(x)$: variables discretas pueden converger a una continua, y densidades altamente oscilatorias como $f_n(x) = 1 - \cos(2n\pi x)$ en $(0,1)$ cumplen $F_n(x) = x - \frac{\sin(2n\pi x)}{2n\pi} \to x$ ($U(0,1)$) sin que $f_n(x)$ converja.
\end{warnbox}

\begin{thmbox}[frametitle={Teorema de Slutsky y estrategia de multiplicar y dividir}]
Si $X_n \dto X$ y $Y_n \pto c$ (equivalentemente $Y_n \dto c$, donde $c \in \R$ es \textbf{constante}), entonces:
\[
X_n \pm Y_n \dto X \pm c, \qquad X_n Y_n \dto c X, \qquad \frac{X_n}{Y_n} \dto \frac{X}{c} \quad (c \ne 0).
\]
\textbf{Estrategia práctica (multiplicar y dividir por el parámetro verdadero $\sigma$ o $\theta$):}
\[
\text{Estadístico} = \underbrace{\frac{\text{término del TLC}}{\text{escala verdadera}}}_{\dto N(0,1)} \times \underbrace{\frac{\text{escala verdadera}}{\text{escala estimada}}}_{\pto 1 \text{ (o } c\text{)}}.
\]
\textit{Ejemplo (TLC estudentizado):} $\dfrac{\bar X_n - \mu}{S/\sqrt{n}} = \dfrac{\bar X_n - \mu}{\sigma/\sqrt{n}} \cdot \dfrac{\sigma}{S} \dto 1 \cdot N(0,1) = N(0,1)$ porque $S^2 \pto \sigma^2 \implies \sigma/S \pto 1$.
\end{thmbox}

\begin{thmbox}[frametitle={Método Delta de 2.\textsuperscript{o} orden (caso degenerado $g'(\mu) = 0$)}]
Si $\sqrt{n}(Y_n - \mu) \dto N(0, \sigma^2)$ ($\sigma^2 > 0$) y $g$ es dos veces derivable en $\mu$ con $g'(\mu) = 0$ y $g''(\mu) \ne 0$, entonces $\sqrt{n}\big(g(Y_n) - g(\mu)\big) \pto 0$ y el reescalado correcto es $n$:
\[
n\big(g(Y_n) - g(\mu)\big) \dto \frac{1}{2}\,g''(\mu)\,\sigma^2\,\chi^2_{(1)}.
\]
\textbf{Ejemplo (varianza muestral Bernoulli):} para $X_i \iid \text{Ber}(p)$ y $\hat\sigma^2 = \bar X(1 - \bar X)$ ($g(x)=x(1-x)$), si $p \ne 1/2$ vale Delta 1.\textsuperscript{er} orden ($\sqrt{n}(\hat\sigma^2 - p(1-p)) \dto N(0, p(1-p)(1-2p)^2)$); en $p = 1/2$, $g'(1/2) = 0$, $g''(1/2) = -2$ y $\sigma^2 = 1/4$:
\[
n\left(\hat\sigma^2 - \frac{1}{4}\right) \dto -\frac{1}{4}\chi^2_{(1)} \qquad\Longleftrightarrow\qquad 4n\left(\frac{1}{4} - \hat\sigma^2\right) \dto \chi^2_{(1)}.
\]
\end{thmbox}

\begin{thmbox}[frametitle={Método Delta Multivariado}]
Sea $\mathbf{Y}_n \in \R^k$ tal que $\sqrt{n}(\mathbf{Y}_n - \boldsymbol{\mu}) \dto N_k(\mathbf{0}, \boldsymbol{\Sigma})$ (para promedios muestrales $\mathbf{Y}_n = \bar{\mathbf{X}}_n$, $\Sigma_{ij} = \Cov(X_{1i}, X_{1j})$). Si $g: \R^k \to \R$ es diferenciable en $\boldsymbol{\mu}$ con $\nabla g(\boldsymbol{\mu}) = \nabla_{\boldsymbol{\mu}} \ne \mathbf{0}$ y $\nabla_{\boldsymbol{\mu}}^\top \boldsymbol{\Sigma} \nabla_{\boldsymbol{\mu}} > 0$, entonces:
\[
\sqrt{n}\big(g(\mathbf{Y}_n) - g(\boldsymbol{\mu})\big) \dto N\!\left(0,\; \nabla_{\boldsymbol{\mu}}^\top \boldsymbol{\Sigma} \nabla_{\boldsymbol{\mu}}\right).
\]
\begin{itemize}
  \item \textbf{Cociente de medias $\bar X / \bar Y$ ($\mu_Y \ne 0$):} con $g(x,y)=x/y$, $\nabla_{\boldsymbol{\mu}} = \big(\frac{1}{\mu_Y},\; -\frac{\mu_X}{\mu_Y^2}\big)^\top$, resulta $\nabla_{\boldsymbol{\mu}}^\top \boldsymbol{\Sigma} \nabla_{\boldsymbol{\mu}} = \frac{\sigma_X^2}{\mu_Y^2} - \frac{2\mu_X\Cov(X,Y)}{\mu_Y^3} + \frac{\mu_X^2\sigma_Y^2}{\mu_Y^4}$.
  \item \textbf{Modelo de Laplace ($f(x;\sigma) = \frac{1}{2\sigma}e^{-|x|/\sigma}$):} para $\mathbf{Y}_n = \big(\frac{1}{n}\sum |X_i|, \bar X\big)^\top$, se tiene $\E[|X_1|]=\sigma$, $\E[X_1]=0$, $\Var(|X_1|)=\sigma^2$, $\Var(X_1)=2\sigma^2$ y por simetría $\Cov(|X_1|, X_1) = 0$ ($\boldsymbol{\Sigma} = \operatorname{diag}(\sigma^2, 2\sigma^2)$).
\end{itemize}
\end{thmbox}

\section{Leyes Débiles y Fuertes de los Grandes Números \textmd{(Clase 13)}}

\begin{thmbox}[frametitle={Leyes Débiles de los Grandes Números (LDGN · convergencia en probabilidad $\pto$)}]
\begin{itemize}
  \item \textbf{LDGN de Bernoulli:} si $X_1, X_2, \dots \iid \text{Ber}(p)$ y $S_n = \sum_{i=1}^n X_i$, entonces $\hat p_n = \dfrac{S_n}{n} \pto p$.
  \item \textbf{LDGN de Khintchine:} si $X_1, X_2, \dots \iid$ con media finita $\E[|X_1|] < \infty$ y $\E[X_1] = \mu$ (\textbf{sin requerir $\sigma^2 < \infty$}), entonces $\bar X_n = \dfrac{S_n}{n} \pto \mu$.
  \item \textbf{LDGN de Chebyshev (no requiere i.i.d.\ ni independencia):} para una sucesión cualquiera $\{X_n\}$ con $S_n = \sum_{i=1}^n X_i$:
  \[
  \lim_{n\to\infty} \frac{\Var(S_n)}{n^2} = 0 \implies \frac{S_n - \E[S_n]}{n} \pto 0.
  \]
  En presencia de dependencia (p.\ ej.\ rachas $k$-dependientes), recuerda incluir las covarianzas cruzadas:
  \[
  \Var(S_n) = \sum_{i=1}^n \Var(X_i) + 2\sum_{1 \le i < j \le n} \Cov(X_i, X_j).
  \]
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Leyes Fuertes de los Grandes Números (LFGN · convergencia casi segura $\dcs$)}]
\begin{itemize}
  \item \textbf{LFGN de Borel:} si $X_1, X_2, \dots \iid \text{Ber}(p)$, entonces $\dfrac{S_n}{n} \dcs p$.
  \item \textbf{LFGN de Kolmogorov (i.i.d.):} si $X_1, X_2, \dots \iid$, entonces $\E[|X_1|] < \infty \iff \bar X_n = \dfrac{S_n}{n} \dcs \mu = \E[X_1]$.
  \item \textbf{LFGN de Kolmogorov (independientes no i.d.):} si $X_1, X_2, \dots$ son \textbf{independientes} con varianzas finitas y
  \[
  \sum_{n=1}^\infty \frac{\Var(X_n)}{n^2} < \infty \implies \frac{S_n - \E[S_n]}{n} \dcs 0.
  \]
\end{itemize}
\end{thmbox}

\begin{notebox}[frametitle={Promedios no lineales (receta de Sesión 2)}]
Reescribe el estadístico en función de promedios $\frac{1}{n}\sum_{i=1}^n h(X_i)$, comprueba $\E[|h(X_1)|] < \infty$, aplica la LGN y concluye por mapeo continuo:
\begin{itemize}
  \item \textbf{Media geométrica} ($X_i > 0$ i.i.d., $\E[|\ln X_1|] < \infty$):
  \[
  G_n = \left(\prod_{i=1}^n X_i\right)^{1/n} = \exp\!\left(\frac{1}{n}\sum_{i=1}^n \ln X_i\right) \dcs e^{\E[\ln X_1]} \qquad \big(\text{ej. } U(0,1) \implies e^{-1}\big).
  \]
  \item \textbf{Media armónica} ($X_i > 0$ i.i.d., $\E[X_1^{-1}] < \infty$):
  \[
  H_n = \left(\frac{1}{n}\sum_{i=1}^n X_i^{-1}\right)^{-1} \dcs \big(\E[X_1^{-1}]\big)^{-1} \qquad \big(\text{ej. } f(x)=3x^2,\ 0 < x < 1 \implies 2/3\big).
  \]
\end{itemize}
\end{notebox}

\section{TLC generalizado, Pivotes Asintóticos y Teoría del Valor Extremo \textmd{(Clase 14)}}

\begin{thmbox}[frametitle={Teoremas del Límite Central (De Moivre--Laplace, Lindeberg--Lévy y no i.d.)}]
\begin{itemize}
  \item \textbf{De Moivre--Laplace:} si $S_n \sim \text{Bin}(n,p)$ con $0 < p < 1$, entonces $\dfrac{S_n - np}{\sqrt{np(1-p)}} \dto N(0,1)$.
  \item \textbf{Lindeberg--Lévy (i.i.d.):} si $X_1, X_2, \dots \iid$ con $\E[X_1]=\mu$ y $\Var(X_1)=\sigma^2 \in (0,\infty)$, entonces $\dfrac{S_n - n\mu}{\sigma\sqrt{n}} = \dfrac{\sqrt{n}(\bar X_n - \mu)}{\sigma} \dto N(0,1)$.
  \item \textbf{TLC de Liapunov--Lindeberg (independientes no i.d.):} sean $X_1, X_2, \dots$ \textbf{independientes} con $\mu_i = \E[X_i]$, $\sigma_i^2 = \Var(X_i)$, $\E[S_n] = \sum_{i=1}^n \mu_i$ y $s_n^2 = \Var(S_n) = \sum_{i=1}^n \sigma_i^2$. En particular, si están uniformemente acotadas ($|X_i| \le M$ c.s.\ para todo $i$) y $s_n^2 \to \infty$, entonces:
  \[
  \frac{S_n - \E[S_n]}{s_n} \dto N(0,1).
  \]
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Estadísticos con residuo, pivotes asintóticos y consistencia universal de $S^2$}]
\begin{itemize}
  \item \textbf{Estadísticos con residuo:} si $T_n = \sum_{i=1}^n g(X_i) + R_n$ con $X_i \iid$, $\E[g(X_1)] = \xi$, $\Var(g(X_1)) = \nu^2 \in (0,\infty)$ y $R_n/\sqrt{n} \pto 0$, entonces el término dominante fija el TLC por Slutsky:
  \[
  \frac{T_n - n\xi}{\nu\sqrt{n}} \dto N(0,1).
  \]
  \item \textbf{Pivotes asintóticos y Delta encadenado:} dividir por un estimador consistente de escala elimina el parámetro desconocido (ej.\ en $U(-\theta,\theta)$ con $Y_n = X_{(n)} \pto \theta$, $\frac{S_n}{\sqrt{n}\,Y_n} \dto N(0, 1/3)$); además, la salida asintótica de un Método Delta puede encadenarse como entrada de otro.
  \item \textbf{Consistencia universal de $S^2$:} para toda población i.i.d.\ con $\E[X_1^2] < \infty$ (sin exigir 4.\textsuperscript{o} momento), por Jensen/Cauchy--Schwarz $\E[|X_1|] \le \sqrt{\E[X_1^2]} < \infty$; reescribiendo $S^2 = \frac{n}{n-1}\big(\frac{1}{n}\sum_{i=1}^n X_i^2 - \bar X_n^2\big)$ y aplicando LFGN se obtiene $S^2 \dcs \sigma^2$ (y $S^2 \pto \sigma^2$).
\end{itemize}
\end{thmbox}

\begin{thmbox}[frametitle={Teoría del Valor Extremo (TVE · Fisher--Tippett--Gnedenko)}]
Sea $X_{(n)} = \max_{1\le i\le n} X_i$ de $X_i \iid$ con FDA $F_X$ ($F_{X_{(n)}}(x) = [F_X(x)]^n$). Si existen sucesiones $a_n \in \R$ y $b_n > 0$ tales que
\[
\frac{X_{(n)} - a_n}{b_n} \dto Y
\]
con distribución no degenerada, entonces $Y$ pertenece necesariamente a una de las \textbf{3 familias (GEV)}:
\begin{enumerate}
  \item \textbf{Tipo 1 · Gumbel} $\text{Gumb}(a,b)$ ($\xi = 0$, colas exponenciales como Exp, Normal, Gamma):
  \[
  G_1(z) = \exp\!\left\{-\exp\!\left[-\frac{z - a}{b}\right]\right\}, \qquad z \in \R \;\;(b > 0).
  \]
  \textit{Caso estándar $\text{Gumb}(0,1)$:} $G_1(z) = e^{-e^{-z}}$, densidad $f(z) = e^{-z-e^{-z}}$, $\E[Z] = \gamma \approx 0.5772$, $\Var(Z) = \pi^2/6$. Para $X_i \iid \text{Exp}(1)$: $X_{(n)} - \ln n \dto \text{Gumb}(0,1)$.
  \item \textbf{Tipo 2 · Fréchet} ($\xi > 0$, colas pesadas / polinomiales como Pareto, Cauchy, $t$):
  \[
  G_2(z) = \exp\!\big\{-(z - a)^{-\alpha}\big\}\,\mathbf{1}_{\{z > a\}}, \qquad \alpha > 0.
  \]
  \item \textbf{Tipo 3 · Weibull} ($\xi < 0$, soporte acotado por arriba como $U(0,1)$, Beta):
  \[
  G_3(z) = \exp\!\big\{-(a - z)^{\alpha}\big\}\,\mathbf{1}_{\{z \le a\}} + \mathbf{1}_{\{z > a\}}, \qquad \alpha > 0.
  \]
\end{enumerate}
\end{thmbox}

\section{Suficiencia y Factorización de Fisher--Neyman \textmd{(Clase 15)}}

\begin{defbox}[frametitle={Definición · Estadístico suficiente}]
Sea $X_1, \dots, X_n$ una muestra aleatoria de una población con parámetro desconocido $\theta$. El estadístico $T = T(X_1, \dots, X_n)$ es \textbf{suficiente para $\theta$} si la distribución condicional de $\mathbf{X} = (X_1, \dots, X_n)$ dado $T = t$,
\[
\Prob(\mathbf{X} = \mathbf{x} \mid T = t) = \Prob(X_1 = x_1, \dots, X_n = x_n \mid T = t),
\]
\textbf{no depende de $\theta$} (en discretas, el cociente $L(\theta \mid \mathbf{x}) / p_T(t;\theta)$ evaluado en $T(\mathbf{x}) = t$ es libre de $\theta$).

Además, si $T$ es suficiente para $\theta$ y $T = \psi(U)$ para alguna función $\psi$ (en particular, si $U$ es una transformación biyectiva de $T$), entonces $U$ también es suficiente para $\theta$ (ej.\ $\sum_{i=1}^n X_i \iff \bar X$).
\end{defbox}

\begin{thmbox}[frametitle={Teorema de Factorización de Fisher--Neyman}]
Un estadístico $T = T(X_1, \dots, X_n)$ es suficiente para $\theta$ si y solo si existen funciones no negativas $g\big(T(\mathbf{x}),\,\theta\big) \ge 0$ (que depende de $\mathbf{x}$ únicamente a través de $T(\mathbf{x})$) y $h(\mathbf{x}) \ge 0$ (independiente de $\theta$) tales que:
\[
L(\theta \mid \mathbf{x}) = g\big(T(\mathbf{x}),\,\theta\big) \times h(\mathbf{x}).
\]
\begin{itemize}
  \item \textbf{Soporte dependiente de $\theta$:} toda función indicadora que involucre a $\theta$ debe quedar absorbida dentro de $g\big(T(\mathbf{x}),\theta\big)$.
  \item \textbf{Conexión con el EMV:} si existe un EMV único $\hat\theta_{\text{MV}}$, maximizar $L(\theta \mid \mathbf{x}) = g(T(\mathbf{x}),\theta)\,h(\mathbf{x})$ respecto a $\theta$ equivale a maximizar $g(T(\mathbf{x}),\theta)$, por lo que $\hat\theta_{\text{MV}}$ es necesariamente función de cualquier estadístico suficiente $T(\mathbf{X})$.
\end{itemize}
\end{thmbox}

\section{Suficiencia conjunta y Familia Exponencial \textmd{(Clase 16)}}

\begin{thmbox}[frametitle={Suficiencia conjunta y Criterio de Factorización multiparamétrico}]
Cuando el modelo depende de un vector de parámetros $\boldsymbol{\theta} = (\theta_1, \dots, \theta_m)$, el estadístico $r$-dimensional $\mathbf{T}(\mathbf{X}) = \big(T_1(\mathbf{X}), \dots, T_r(\mathbf{X})\big)$ es \textbf{conjuntamente suficiente para $\boldsymbol{\theta}$} si y solo si la verosimilitud puede escribirse como:
\[
L(\boldsymbol{\theta} \mid \mathbf{x}) = \prod_{i=1}^n f(x_i \mid \boldsymbol{\theta}) = g\big(T_1(\mathbf{x}), \dots, T_r(\mathbf{x}),\,\boldsymbol{\theta}\big)\,h(\mathbf{x}),
\]
donde $g(\mathbf{T}(\mathbf{x}), \boldsymbol{\theta}) \ge 0$ depende de $\mathbf{x}$ únicamente a través de $\mathbf{T}(\mathbf{x})$ y $h(\mathbf{x}) \ge 0$ no depende de $\boldsymbol{\theta}$.
\begin{itemize}
  \item \textbf{Equivalencia biyectiva vectorial:} si $\mathbf{T}$ es conjuntamente suficiente y puede recuperarse como función de $\mathbf{U} = \boldsymbol{\phi}(\mathbf{T})$, entonces $\mathbf{U}$ también lo es (ej.\ $\big(\sum X_i, \sum X_i^2\big) \iff (\bar X, S^2)$ para $n \ge 2$; $\big(\prod X_i, \sum X_i\big) \iff \big(\sum \ln X_i, \sum X_i\big)$ si $X_i > 0$).
  \item \textbf{Soporte dependiente multiparamétrico:} en $U(\theta_1, \theta_2)$ ($\theta_1 < \theta_2$) es conjuntamente suficiente $\mathbf{T}(\mathbf{X}) = (X_{(1)}, X_{(n)})$; en $\text{Pareto}(\alpha, \beta)$ con ambos desconocidos, $\big(X_{(1)}, \sum \ln X_i\big)$.
\end{itemize}
\end{thmbox}

\begin{defbox}[frametitle={Familia Exponencial (FE) 1D y $k$-dimensional}]
Una distribución pertenece a la \textbf{familia exponencial (FE) unidimensional} si su fdp (o fp) puede escribirse como:
\[
f(x \mid \theta) = \exp\big\{c(\theta)T(x) + d(\theta) + S(x)\big\}, \qquad x \in A \;\;(\text{soporte } A \text{ independiente de }\theta),
\]
y a la \textbf{FE $k$-dimensional} para el vector de parámetros $\boldsymbol{\theta} = (\theta_1, \dots, \theta_m)$ si:
\[
f(x \mid \boldsymbol{\theta}) = \exp\left\{\sum_{j=1}^k c_j(\boldsymbol{\theta})T_j(x) + d(\boldsymbol{\theta}) + S(x)\right\}, \qquad x \in A \;\;(\text{fijo, libre de }\boldsymbol{\theta}).
\]
\begin{itemize}
  \item \textbf{Identidades de momentos (1D, $c'(\theta) \ne 0$):} derivando $\int_A f(x \mid \theta)\,dx = 1$ (o $\sum_{x \in A}$) respecto a $\theta$ se obtiene:
  \[
  \E\big[T(X)\big] = -\frac{d'(\theta)}{c'(\theta)}, \qquad \Var\big(T(X)\big) = \frac{d'(\theta)c''(\theta) - c'(\theta)d''(\theta)}{[c'(\theta)]^3}.
  \]
  \item \textbf{Cerradura bajo muestras i.i.d.\ y suficiencia directa:} para $X_1, \dots, X_n \iid$ de una FE $k$-dimensional,
  \[
  f(\mathbf{x} \mid \boldsymbol{\theta}) = \exp\left\{\sum_{j=1}^k c_j(\boldsymbol{\theta})\sum_{i=1}^n T_j(x_i) + n\,d(\boldsymbol{\theta}) + \sum_{i=1}^n S(x_i)\right\}, \qquad \mathbf{x} \in A^n,
  \]
  que permanece en la FE, y por Fisher--Neyman el vector $\mathbf{T}(\mathbf{X}) = \big(\sum_{i=1}^n T_1(X_i), \dots, \sum_{i=1}^n T_k(X_i)\big)$ es \textbf{conjuntamente suficiente para $\boldsymbol{\theta}$} (en 1D, $T(\mathbf{X}) = \sum_{i=1}^n T(X_i)$).
  \item \textbf{Modelos fuera de la FE:} aquellos cuyo soporte depende del parámetro ($U(0,\theta)$, $U(\theta_1,\theta_2)$, Exp desplazada) o donde $\ln f(x\mid\theta)$ no separa $x$ y $\theta$ en suma finita $\sum c_j(\theta)T_j(x)$ (como Cauchy).
\end{itemize}
\end{defbox}

\begin{thmbox}[frametitle={Ecuación general del EMV en la Familia Exponencial}]
En una familia exponencial regular (con jacobiano $[\partial c_j / \partial \theta_m]$ invertible y máximo en el interior del espacio paramétrico), igualar el gradiente de la log-verosimilitud a cero equivale al sistema:
\[
\E_{\boldsymbol{\theta}}\big[T_j(\mathbf{X})\big] = T_j(\mathbf{X}), \qquad j = 1, \dots, k \qquad\Big(\text{en 1D con } c'(\theta) \ne 0:\; \E_\theta\big[T(\mathbf{X})\big] = T(\mathbf{X})\Big).
\]
\end{thmbox}

\section{Estimadores MVUE y Teorema de Rao--Blackwell \textmd{(Clase 17)}}

\begin{defbox}[frametitle={Definición · Estimador MVUE / UMVUE}]
Un estadístico $\hat\theta^* = \hat\theta^*(X_1, \dots, X_n)$ es un \textbf{estimador insesgado de varianza mínima uniforme (MVUE / UMVUE)} para $\theta$ (análogamente para $\tau(\theta)$) si:
\begin{enumerate}
  \item \textbf{Insesgamiento:} $\E_\theta[\hat\theta^*] = \theta$ para todo $\theta \in \Omega$.
  \item \textbf{Varianza mínima uniforme:} $\Var_\theta(\hat\theta^*) \le \Var_\theta(\hat\theta)$ para todo $\theta \in \Omega$ y cualquier otro estimador insesgado $\hat\theta$ de $\theta$.
\end{enumerate}
\end{defbox}

\begin{thmbox}[frametitle={Teorema de Rao--Blackwell}]
Sea $\hat\theta$ un estimador insesgado para $\theta$ con $\Var(\hat\theta) < \infty$, y sea $T = T(X_1, \dots, X_n)$ un estadístico suficiente para $\theta$. Definimos el nuevo estimador:
\[
\hat\theta^* = \E[\hat\theta \mid T].
\]
Entonces $\hat\theta^*$ es un estadístico válido (libre de $\theta$), es \textbf{insesgado} ($\E_\theta[\hat\theta^*] = \E_\theta[\E[\hat\theta \mid T]] = \theta$) y satisface:
\[
\Var(\hat\theta^*) = \Var(\hat\theta) - \E\big[\Var(\hat\theta \mid T)\big] \le \Var(\hat\theta)
\]
(con igualdad s.s.\ cuando $\hat\theta$ ya es función de $T$ c.s.; el mismo teorema aplica a $\tau(\theta)$ tomando $\hat\tau^* = \E[\hat\tau \mid T]$).
\end{thmbox}

\begin{notebox}[frametitle={Teorema de Lehmann--Scheffé (completitud y dos recetas prácticas)}]
Si el estadístico suficiente $T$ es además \textbf{completo} ($\E_\theta[h(T)] = 0\;\forall\theta \implies h(T) = 0$ c.s., como ocurre en las FE de rango completo y en $U(0,\theta)$), existe un \textbf{único} estimador insesgado función de $T$. Por tanto, $\hat\theta^* = \E[\hat\theta \mid T]$ es único y es el \textbf{MVUE}.
\begin{itemize}
  \item \textbf{Receta 1 (vía directa):} hallar $T$ suficiente completo y ajustar una función $\phi(T)$ tal que $\E_\theta[\phi(T)] = \tau(\theta)$.
  \item \textbf{Receta 2 (condicionamiento):} partir de un insesgado elemental $\hat\tau$ y calcular explícitamente $\E[\hat\tau \mid T]$.
\end{itemize}
\end{notebox}

\begin{thmbox}[frametitle={EMV, suficiencia y comportamiento en grandes muestras}]
\begin{itemize}
  \item \textbf{Relación EMV--suficiencia:} si $T$ es suficiente para $\theta$ y el EMV $\hat\theta$ existe y es único, entonces $\hat\theta$ es necesariamente \textbf{función de $T$} (maximizar $L(\theta \mid \mathbf{x}) = g(T(\mathbf{x}),\theta)\,h(\mathbf{x})$ equivale a maximizar $g(T(\mathbf{x}),\theta)$).
  \item \textbf{Grandes muestras (normalidad y eficiencia asintótica):} bajo regularidad ($0 < I_F(\theta) < \infty$) y para $g$ derivable con $g'(\theta) \ne 0$:
  \[
  \sqrt{n}(\hat\theta - \theta) \dto N\!\left(0,\, \frac{1}{I_F(\theta)}\right), \qquad \sqrt{n}\big(g(\hat\theta) - g(\theta)\big) \dto N\!\left(0,\, \frac{[g'(\theta)]^2}{I_F(\theta)}\right).
  \]
  Los EMV son \textbf{asintóticamente insesgados y eficientes} (alcanzan la cota de Cramér--Rao $\frac{1}{nI_F(\theta)}$ y $\frac{[g'(\theta)]^2}{nI_F(\theta)}$).
  \item \textbf{Sustitución consistente:} la convergencia estandarizada se conserva al reemplazar $\theta$ por cualquier estimador consistente $\tilde\theta \pto \theta$ (si $I_F, g'$ son continuas y no nulas):
  \[
  \sqrt{n\,I_F(\tilde\theta)}(\hat\theta - \theta) \dto N(0, 1), \qquad \frac{\sqrt{n\,I_F(\tilde\theta)}}{|g'(\tilde\theta)|}\big(g(\hat\theta) - g(\theta)\big) \dto N(0, 1).
  \]
\end{itemize}
\end{thmbox}

\section{Cómo leer las tablas y los valores críticos}

\begin{notebox}[frametitle={La regla que evita todos los líos de colas}]
Las tablas ESPOL dan el valor con \textbf{área a la IZQUIERDA} $q$:
$\Prob(X\le x)=q$. En clase se escribe el valor crítico con \textbf{área
$\alpha$ a la DERECHA}. Entonces:
\[
\text{valor crítico con cola derecha }\alpha
\;=\; \text{valor de la tabla en la columna } q=1-\alpha .
\]
El \textbf{percentil 100$p$} es el $x_p$ con área $p$ a la izquierda; el valor
crítico al $5\%$ es el \textbf{percentil 95}.
\end{notebox}

$\chi^2$ & \rr fila $\nu$, columna $p$: $\Prob(\chi^2_{(\nu)}\le x)=p$ & \rr $\chi^2_{(\nu),\alpha}$ = columna $1-\alpha$\\
$t$ & \rr fila $\nu$, columna $p$: $\Prob(t_{(\nu)}\le x)=p$ (sólo $p\ge0.6$) & \rr $t_{(\nu),\alpha}$ = columna $1-\alpha$; por simetría $\Prob(|T|\le c)=2\Prob(T\le c)-1$\\
$F$ & \rr bloque $\nu_2$ (denominador), fila $q$, columna $\nu_1$ (numerador) & \rr $F_{(\nu_1,\nu_2,\alpha)}$ = fila $q=1-\alpha$, columna $\nu_1$\\

  \item Cola izquierda (no hay fila $q<0.5$):
        $F_{(\nu_1,\nu_2,1-\alpha)}=1/F_{(\nu_2,\nu_1,\alpha)}$.
\end{itemize}
